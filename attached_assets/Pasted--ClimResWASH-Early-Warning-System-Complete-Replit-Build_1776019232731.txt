# ClimResWASH Early Warning System — Complete Replit Build Guide

## What You're Building

A Python Flask backend with two regression models that integrate into your existing climreswash.in website:

- **Layer 1 (Short-term):** Predicts flood / heatwave / drought probability for the next 1-7 days per district → triggers quick protective measures for existing WASH infrastructure
- **Layer 2 (Long-term):** Scores each district's chronic proneness to hazards → recommends which toilet and liquid waste technology to install

```python
def parse_warning_to_hazards(warning_data):
    """
    Convert IMD warning codes to hazard probabilities.
    
    Maps: color code → probability range
    Red (1) → 0.85-0.95
    Orange (2) → 0.55-0.75
    Yellow (3) → 0.25-0.45
    Green (4) → 0.0-0.15
    """
    if not warning_data:
        return {"flood": 0, "heatwave": 0, "drought": 0}
    
    hazards = {"flood": 0, "heatwave": 0, "drought": 0}
    
    # Map warning codes to hazard types
    flood_codes = {2, 3, 16, 17}     # Heavy/Very Heavy/Extreme rain, snow
    heat_codes = {9, 10, 11}          # Heat wave, hot day, warm night
    drought_indicators = {"D", "LD", "NR"}  # Deficient rainfall categories
    
    # Color to probability mapping
    color_prob = {1: 0.9, 2: 0.65, 3: 0.35, 4: 0.05}
    
    for record in (warning_data if isinstance(warning_data, list) 
                   else [warning_data]):
        for day in range(1, 6):
            day_key = f"Day_{day}"
            color_key = f"Day{day}_Color"
            
            if day_key not in record or color_key not in record:
                continue
                
            codes = str(record[day_key]).split(",")
            color = int(record.get(color_key, 4))
            prob = color_prob.get(color, 0.05)
            
            for code in codes:
                code = code.strip()
                if not code.isdigit():
                    continue
                code = int(code)
                
                if code in flood_codes:
                    hazards["flood"] = max(hazards["flood"], prob)
                elif code in heat_codes:
                    hazards["heatwave"] = max(hazards["heatwave"], prob)
    
    return hazards
```

```python
"""
Open-Meteo API — free, no auth, no rate limits for non-commercial use.
Use as primary source until IMD IP is whitelisted.
Docs: https://open-meteo.com/en/docs
"""
import requests
from config import OPENMETEO_FORECAST, OPENMETEO_HISTORICAL, OPENMETEO_FLOOD


def fetch_forecast(lat, lon, days=7):
    """
    Fetch weather forecast for a location.
    Returns hourly/daily temperature, rainfall, humidity, wind.
    
    No API key needed. No registration needed.
    """
    params = {
        "latitude": lat,
        "longitude": lon,
        "daily": ",".join([
            "temperature_2m_max",
            "temperature_2m_min",
            "precipitation_sum",
            "rain_sum",
            "precipitation_probability_max",
            "windspeed_10m_max",
            "relative_humidity_2m_mean",  
            "et0_fao_evapotranspiration",
        ]),
        "forecast_days": days,
        "timezone": "Asia/Kolkata",
    }
    
    try:
        resp = requests.get(OPENMETEO_FORECAST, params=params, timeout=15)
        resp.raise_for_status()
        data = resp.json()
        return data.get("daily", {})
    except Exception as e:
        print(f"[Open-Meteo] Forecast failed: {e}")
        return None


def fetch_historical(lat, lon, start_date, end_date):
    """
    Fetch historical weather data (goes back 80+ years).
    Use for training long-term model and calculating normals.
    
    start_date/end_date format: "YYYY-MM-DD"
    """
    params = {
        "latitude": lat,
        "longitude": lon,
        "start_date": start_date,
        "end_date": end_date,
        "daily": ",".join([
            "temperature_2m_max",
            "temperature_2m_min",
            "precipitation_sum",
            "rain_sum",
            "windspeed_10m_max",
            "relative_humidity_2m_mean",
        ]),
        "timezone": "Asia/Kolkata",
    }
    
    try:
        resp = requests.get(OPENMETEO_HISTORICAL, params=params, timeout=30)
        resp.raise_for_status()
        return resp.json().get("daily", {})
    except Exception as e:
        print(f"[Open-Meteo] Historical failed: {e}")
        return None


def fetch_flood_forecast(lat, lon):
    """
    Fetch river discharge forecast from Open-Meteo Flood API.
    Uses GloFAS (Global Flood Awareness System) data.
    """
    params = {
        "latitude": lat,
        "longitude": lon,
        "daily": "river_discharge",
        "forecast_days": 7,
    }
    
    try:
        resp = requests.get(OPENMETEO_FLOOD, params=params, timeout=15)
        resp.raise_for_status()
        return resp.json().get("daily", {})
    except Exception as e:
        print(f"[Open-Meteo] Flood forecast failed: {e}")
        return None


def compute_hazard_from_forecast(forecast_data):
    """
    Convert Open-Meteo forecast into hazard probabilities.
    This is the fallback logic when IMD is unavailable.
    """
    if not forecast_data:
        return {"flood": 0.0, "heatwave": 0.0, "drought": 0.0}
    
    temps_max = forecast_data.get("temperature_2m_max", [])
    precip = forecast_data.get("precipitation_sum", [])
    humidity = forecast_data.get("relative_humidity_2m_mean", [])
    
    # Flood probability
    max_rain = max(precip) if precip else 0
    avg_rain = sum(precip) / len(precip) if precip else 0
    flood_prob = min(1.0, max_rain / 150 * 0.5 + avg_rain / 80 * 0.3)
    
    # Heatwave probability
    max_temp = max(temps_max) if temps_max else 0
    days_above_40 = sum(1 for t in temps_max if t and t >= 40)
    heat_prob = min(1.0, 
        (max(0, max_temp - 38) / 12) * 0.5 + 
        (days_above_40 / max(len(temps_max), 1)) * 0.5
    )
    
    # Drought probability
    dry_days = sum(1 for p in precip if p is not None and p < 1)
    total_rain = sum(p for p in precip if p is not None)
    drought_prob = min(1.0,
        (dry_days / max(len(precip), 1)) * 0.4 +
        max(0, 1 - total_rain / 20) * 0.3 +
        (max(0, max_temp - 35) / 15) * 0.3
    )
    
    return {
        "flood": round(flood_prob, 3),
        "heatwave": round(heat_prob, 3),
        "drought": round(drought_prob, 3),
    }
```

### 5c. CGWB Groundwater Fetcher

Create `data/cgwb_fetcher.py`:

```python
"""
CGWB Groundwater data.
The CSV dataset has 22,965 observation wells measured 4 times/year.
Download once and cache locally.

Sources:
- India Data Portal CSV: district-level depth-to-water-level
- India-WRIS: https://indiawris.gov.in/wris/#/
- CGWB portal: https://gwdata.cgwb.gov.in/
"""
import os
import csv
import requests
from config import CGWB_CSV_URL


CACHE_FILE = "data/cgwb_cache.csv"


def download_cgwb_data():
    """
    Download CGWB groundwater CSV (one-time, ~5MB).
    Contains columns: State, District, Well_Code, Lat, Long,
    Season, Year, Depth_to_Water_Level (meters)
    """
    if os.path.exists(CACHE_FILE):
        print("[CGWB] Using cached data")
        return True
    
    print("[CGWB] Downloading groundwater dataset...")
    try:
        resp = requests.get(CGWB_CSV_URL, timeout=60)
        resp.raise_for_status()
        os.makedirs("data", exist_ok=True)
        with open(CACHE_FILE, "w") as f:
            f.write(resp.text)
        print(f"[CGWB] Downloaded {len(resp.text)} bytes")
        return True
    except Exception as e:
        print(f"[CGWB] Download failed: {e}")
        return False


def get_district_gw_depth(district_name):
    """
    Get latest groundwater depth for a district.
    Returns depth in meters (lower = shallower = higher flood risk).
    """
    if not os.path.exists(CACHE_FILE):
        download_cgwb_data()
    
    if not os.path.exists(CACHE_FILE):
        return None
    
    district_upper = district_name.upper()
    latest_depth = None
    latest_year = 0
    
    try:
        with open(CACHE_FILE, "r") as f:
            reader = csv.DictReader(f)
            for row in reader:
                if district_upper in str(row.get("District", "")).upper():
                    year = int(row.get("Year", 0) or 0)
                    depth = row.get("Depth_to_Water_Level", "")
                    if year > latest_year and depth:
                        try:
                            latest_depth = float(depth)
                            latest_year = year
                        except ValueError:
                            continue
    except Exception as e:
        print(f"[CGWB] Parse error: {e}")
    
    return latest_depth


def get_district_gw_trend(district_name, years=20):
    """
    Calculate long-term groundwater trend for a district.
    Returns: trend (m/year), positive = deepening (bad),
    negative = rising (flood risk if near surface).
    """
    if not os.path.exists(CACHE_FILE):
        download_cgwb_data()
    
    if not os.path.exists(CACHE_FILE):
        return None
    
    district_upper = district_name.upper()
    yearly_depths = {}
    
    try:
        with open(CACHE_FILE, "r") as f:
            reader = csv.DictReader(f)
            for row in reader:
                if district_upper in str(row.get("District", "")).upper():
                    year = int(row.get("Year", 0) or 0)
                    depth = row.get("Depth_to_Water_Level", "")
                    if year > 0 and depth:
                        try:
                            d = float(depth)
                            yearly_depths.setdefault(year, []).append(d)
                        except ValueError:
                            continue
    except Exception:
        return None
    
    if len(yearly_depths) < 3:
        return None
    
    # Simple linear trend: average depth per year
    years_list = sorted(yearly_depths.keys())
    avgs = [(y, sum(yearly_depths[y]) / len(yearly_depths[y])) 
            for y in years_list]
    
    if len(avgs) < 2:
        return None
    
    # Slope calculation
    n = len(avgs)
    sum_x = sum(a[0] for a in avgs)
    sum_y = sum(a[1] for a in avgs)
    sum_xy = sum(a[0] * a[1] for a in avgs)
    sum_x2 = sum(a[0] ** 2 for a in avgs)
    
    denom = n * sum_x2 - sum_x ** 2
    if denom == 0:
        return 0
    
    slope = (n * sum_xy - sum_x * sum_y) / denom
    return round(slope, 3)  # meters per year
```

## Step 6: Prediction Models

### 6a. Short-Term Model (Layer 1)

Create `models/short_term_model.py`:

```python
"""
Layer 1: Short-term hazard prediction (1-7 days ahead).

Predicts probability of flood, heatwave, drought for a district
using current/forecast weather data.

Starts with rule-based model (works immediately, no training needed).
Upgrade to XGBoost when you have 6+ months of labeled data.
"""
import numpy as np
from config import THRESHOLDS


class ShortTermPredictor:
    """
    Rule-based predictor (Phase 1).
    Uses IMD official thresholds + weighted scoring.
    
    Replace with trained XGBoost model in Phase 2 once you have
    labeled data (did flood/heatwave/drought actually occur Y/N).
    """
    
    def predict(self, features):
        """
        Predict hazard probabilities from current conditions.
        
        Args:
            features: dict with keys:
                - rainfall_mm: current/forecast rainfall (mm/day)
                - temperature_c: max temperature (°C)
                - temp_departure_c: departure from normal (°C)
                - humidity_pct: relative humidity (%)
                - wind_speed_kmph: wind speed (km/h)
                - gw_depth_m: groundwater depth (meters)
                - river_level_pct: river level as % of danger level
                - consecutive_dry_days: days without rain
                - rainfall_departure_pct: % departure from normal rainfall
                - imd_warning_color: IMD color code (1=Red to 4=Green)
                - soil_permeability: 1=Low, 2=Medium, 3=High
        
        Returns:
            dict with flood, heatwave, drought probabilities (0-1)
        """
        rain = features.get("rainfall_mm", 0)
        temp = features.get("temperature_c", 30)
        temp_dep = features.get("temp_departure_c", 0)
        humidity = features.get("humidity_pct", 50)
        gw = features.get("gw_depth_m", 10)
        river = features.get("river_level_pct", 50)
        dry_days = features.get("consecutive_dry_days", 0)
        rain_dep = features.get("rainfall_departure_pct", 0)
        soil = features.get("soil_permeability", 2)
        
        # ---- FLOOD PREDICTION ----
        flood_score = 0.0
        
        # Rainfall intensity (biggest factor)
        t = THRESHOLDS["flood"]
        if rain >= t["rainfall_extreme_mm_day"]:
            flood_score += 0.45
        elif rain >= t["rainfall_heavy_mm_day"]:
            flood_score += 0.30
        elif rain >= 64.5:  # IMD "Heavy Rain" threshold
            flood_score += 0.15
        else:
            flood_score += min(0.10, rain / 650)
        
        # River level contribution
        if river >= t["river_level_danger_pct"]:
            flood_score += 0.30
        elif river >= t["river_level_warning_pct"]:
            flood_score += 0.15
        else:
            flood_score += river / 700
        
        # Shallow groundwater amplifies flood risk to pits
        if gw < t["gw_shallow_risk_m"]:
            flood_score += 0.15 * (1 - gw / t["gw_shallow_risk_m"])
        
        # Low soil permeability = water doesn't drain
        flood_score += (1 / soil) * 0.08
        
        # Excess rainfall departure
        if rain_dep > t["rainfall_departure_excess_pct"]:
            flood_score += 0.10
        
        flood_prob = min(0.99, max(0.0, flood_score))
        
        # ---- HEATWAVE PREDICTION ----
        heat_score = 0.0
        t_hw = THRESHOLDS["heatwave"]
        
        # Absolute temperature
        if temp >= t_hw["temp_absolute_severe_c"]:
            heat_score += 0.50
        elif temp >= t_hw["temp_absolute_hw_c"]:
            heat_score += 0.40
        elif temp >= t_hw["temp_plains_c"]:
            heat_score += 0.20
        else:
            heat_score += max(0, (temp - 35) / 30)
        
        # Temperature departure from normal
        if temp_dep >= t_hw["departure_severe_c"]:
            heat_score += 0.30
        elif temp_dep >= t_hw["departure_heatwave_c"]:
            heat_score += 0.20
        elif temp_dep > 2:
            heat_score += 0.08
        
        # Low humidity + high temp = worse heat stress
        if humidity < 30 and temp > 40:
            heat_score += 0.10
        
        # Consecutive dry days amplify heat
        if dry_days > 5:
            heat_score += min(0.10, dry_days / 100)
        
        heat_prob = min(0.99, max(0.0, heat_score))
        
        # ---- DROUGHT PREDICTION ----
        drought_score = 0.0
        t_dr = THRESHOLDS["drought"]
        
        # Rainfall deficit
        if rain_dep <= t_dr["rainfall_large_deficient_pct"]:
            drought_score += 0.40
        elif rain_dep <= t_dr["rainfall_deficient_pct"]:
            drought_score += 0.25
        
        # Consecutive dry days
        if dry_days >= t_dr["consecutive_dry_days"]:
            drought_score += 0.30
        elif dry_days >= 7:
            drought_score += 0.15
        else:
            drought_score += dry_days / 100
        
        # Deep groundwater = water stress
        if gw >= t_dr["gw_deep_stress_m"]:
            drought_score += 0.20
        elif gw >= 10:
            drought_score += 0.10
        
        # Low current rainfall
        if rain < 2:
            drought_score += 0.10
        
        drought_prob = min(0.99, max(0.0, drought_score))
        
        return {
            "flood": round(flood_prob, 3),
            "heatwave": round(heat_prob, 3),
            "drought": round(drought_prob, 3),
            "dominant_hazard": max(
                ["flood", "heatwave", "drought"],
                key=lambda h: {"flood": flood_prob, 
                              "heatwave": heat_prob,
                              "drought": drought_prob}[h]
            ),
        }
    
    def get_risk_level(self, probability):
        """Convert probability to human-readable risk level."""
        if probability >= 0.75:
            return {"level": "CRITICAL", "color": "#E24B4A"}
        elif probability >= 0.50:
            return {"level": "HIGH", "color": "#EF9F27"}
        elif probability >= 0.25:
            return {"level": "MODERATE", "color": "#F0997B"}
        else:
            return {"level": "LOW", "color": "#5DCAA5"}
```

```python
"""
Layer 2: Long-term district proneness scoring.

Scores each district's chronic vulnerability to flood, heatwave,
drought using 10-30 year historical data.

Output feeds into WASH technology recommender.
"""
import numpy as np


class LongTermPronenessModel:
    """
    Scores district-level hazard proneness (0-1) per hazard type.
    Uses historical data aggregates.
    
    Upgrade path:
    Phase 1 (now): Weighted formula using available data
    Phase 2: Train Random Forest on CEEW/NDMA district risk data
    Phase 3: Add CMIP6 climate projections for future proneness
    """
    
    def score_district(self, profile):
        """
        Calculate hazard proneness scores for a district.
        
        Args:
            profile: dict with keys:
                - avg_annual_rainfall_mm: 30-year average
                - flood_events_per_decade: historical count
                - avg_gw_depth_m: average groundwater depth
                - gw_trend_m_per_year: +ve = deepening, -ve = rising
                - soil_permeability: 1=Low, 2=Medium, 3=High
                - elevation_m: average elevation
                - heatwave_days_per_year: average
                - drought_months_per_year: average
                - coastal_district: True/False
                - river_basin: True if in major river floodplain
        
        Returns:
            dict with proneness scores and recommended actions
        """
        rain = profile.get("avg_annual_rainfall_mm", 1100)
        floods = profile.get("flood_events_per_decade", 2)
        gw = profile.get("avg_gw_depth_m", 8)
        gw_trend = profile.get("gw_trend_m_per_year", 0)
        soil = profile.get("soil_permeability", 2)
        elevation = profile.get("elevation_m", 200)
        hw_days = profile.get("heatwave_days_per_year", 10)
        drought_months = profile.get("drought_months_per_year", 1)
        coastal = profile.get("coastal_district", False)
        river_basin = profile.get("river_basin", False)
        
        # ---- FLOOD PRONENESS ----
        flood_score = 0.0
        
        # Historical flood frequency (strongest signal)
        flood_score += min(0.35, floods / 12)
        
        # High rainfall areas
        if rain > 2500:
            flood_score += 0.20
        elif rain > 1500:
            flood_score += 0.12
        elif rain > 1000:
            flood_score += 0.05
        
        # Shallow / rising groundwater
        if gw < 3:
            flood_score += 0.15
        elif gw < 5:
            flood_score += 0.08
        if gw_trend < -0.1:  # Rising water table
            flood_score += 0.08
        
        # Low permeability soil
        flood_score += (1 / soil) * 0.08
        
        # Coastal + river basin
        if coastal:
            flood_score += 0.10
        if river_basin:
            flood_score += 0.10
        
        # Low elevation
        if elevation < 50:
            flood_score += 0.08
        
        flood_prone = min(1.0, flood_score)
        
        # ---- HEATWAVE PRONENESS ----
        heat_score = 0.0
        
        # Historical heatwave frequency
        heat_score += min(0.40, hw_days / 50)
        
        # Arid / semi-arid (low rainfall = more heat)
        if rain < 500:
            heat_score += 0.25
        elif rain < 800:
            heat_score += 0.15
        elif rain < 1200:
            heat_score += 0.05
        
        # Deep groundwater (less evaporative cooling)
        if gw > 20:
            heat_score += 0.10
        elif gw > 10:
            heat_score += 0.05
        
        # Inland, non-coastal (less sea breeze moderation)
        if not coastal:
            heat_score += 0.05
        
        heat_prone = min(1.0, heat_score)
        
        # ---- DROUGHT PRONENESS ----
        drought_score = 0.0
        
        # Historical drought frequency
        drought_score += min(0.35, drought_months / 5)
        
        # Low rainfall
        if rain < 400:
            drought_score += 0.30
        elif rain < 700:
            drought_score += 0.20
        elif rain < 1000:
            drought_score += 0.08
        
        # Deepening groundwater trend
        if gw_trend > 0.3:
            drought_score += 0.15
        elif gw_trend > 0.1:
            drought_score += 0.08
        
        # Already deep groundwater
        if gw > 20:
            drought_score += 0.12
        elif gw > 12:
            drought_score += 0.06
        
        drought_prone = min(1.0, drought_score)
        
        return {
            "flood_proneness": round(flood_prone, 3),
            "heatwave_proneness": round(heat_prone, 3),
            "drought_proneness": round(drought_prone, 3),
            "dominant_hazard": max(
                ["flood", "heatwave", "drought"],
                key=lambda h: {
                    "flood": flood_prone,
                    "heatwave": heat_prone,
                    "drought": drought_prone,
                }[h]
            ),
        }
```

```python
def long_term_recommendation(district_name):
    """
    Long-term proneness scoring + WASH technology recommendation.
    
    Query params (all optional, provide what you have):
      avg_rainfall=1200  (mm/year)
      flood_events=4     (per decade)
      gw_depth=6         (meters)
      soil=2             (1=Low, 2=Medium, 3=High permeability)
      hw_days=12         (heatwave days per year)
      drought_months=1   (per year)
      coastal=false
      river_basin=true
    """
    # Get CGWB data
    gw_depth = get_district_gw_depth(district_name)
    gw_trend = get_district_gw_trend(district_name)
    
    profile = {
        "avg_annual_rainfall_mm": request.args.get("avg_rainfall", 1100, type=float),
        "flood_events_per_decade": request.args.get("flood_events", 2, type=int),
        "avg_gw_depth_m": gw_depth or request.args.get("gw_depth", 8, type=float),
        "gw_trend_m_per_year": gw_trend or 0,
        "soil_permeability": request.args.get("soil", 2, type=int),
        "elevation_m": request.args.get("elevation", 200, type=float),
        "heatwave_days_per_year": request.args.get("hw_days", 10, type=int),
        "drought_months_per_year": request.args.get("drought_months", 1, type=int),
        "coastal_district": request.args.get("coastal", "false").lower() == "true",
        "river_basin": request.args.get("river_basin", "false").lower() == "true",
    }
    
    # Score proneness
    scores = long_term.score_district(profile)
    
    # Get technology recommendations
    recommendations = recommend_technologies(scores)
    
    return jsonify({
        "district": district_name,
        "profile_used": profile,
        "proneness_scores": scores,
        "recommendations": recommendations,
    })


@app.route("/api/predict", methods=["POST"])
def predict_custom():
    """
    POST custom features for short-term prediction.
    Useful for your frontend sliders / what-if analysis.
    
    POST JSON body:
    {
        "rainfall_mm": 85,
        "temperature_c": 42,
        "humidity_pct": 65,
        "gw_depth_m": 3.5,
        "river_level_pct": 75,
        "consecutive_dry_days": 5,
        "rainfall_departure_pct": 40,
        "soil_permeability": 2
    }
    """
    features = request.get_json()
    if not features:
        return jsonify({"error": "Send JSON body with features"}), 400
    
    prediction = short_term.predict(features)
    
    # Attach quick measures
    triggered = {}
    for hazard_type in ["flood", "heatwave", "drought"]:
        prob = prediction.get(hazard_type, 0)
        if hazard_type in QUICK_MEASURES:
            threshold = QUICK_MEASURES[hazard_type]["threshold"]
            if prob >= threshold:
                triggered[hazard_type] = {
                    "probability": prob,
                    "risk_level": short_term.get_risk_level(prob),
                    "measures": QUICK_MEASURES[hazard_type]["measures"],
                }
    
    prediction["triggered_measures"] = triggered
    return jsonify(prediction)


@app.route("/api/quick-measures/<hazard>")
def get_quick_measures(hazard):
    """Get all quick measures for a specific hazard type."""
    if hazard not in QUICK_MEASURES:
        return jsonify({
            "error": f"Unknown hazard: {hazard}",
            "available": list(QUICK_MEASURES.keys())
        }), 404
    return jsonify(QUICK_MEASURES[hazard])


def _build_alert_message(hazards, triggered):
    """Build human-readable alert message."""
    if not triggered:
        return "All hazard levels LOW. No immediate action needed."
    
    parts = []
    for h, data in triggered.items():
        level = data["risk_level"]["level"]
        pct = round(data["probability"] * 100)
        count = len(data["measures"])
        parts.append(f"{h.upper()} risk {level} ({pct}%) — {count} measures activated")
    
    return " | ".join(parts)
```
